cmake_minimum_required(VERSION 3.15)
project(GriffithEnergyBalance LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "The native GUI requires Windows. The griffith_core module is portable C++17.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(griffith_core STATIC griffith_core.cpp)
target_include_directories(griffith_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(Griffith WIN32 griffith_app.cpp)
target_compile_definitions(Griffith PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0601)
target_link_libraries(Griffith PRIVATE griffith_core gdiplus gdi32 user32 comctl32 comdlg32 shell32)
add_executable(test_griffith_core test_griffith_core.cpp)
target_link_libraries(test_griffith_core PRIVATE griffith_core)
if(MSVC)
    foreach(target griffith_core Griffith test_griffith_core)
        target_compile_options(${target} PRIVATE /utf-8 /W4)
    endforeach()
elseif(MINGW)
    foreach(target griffith_core Griffith test_griffith_core)
        target_compile_options(${target} PRIVATE -Wall -Wextra -finput-charset=UTF-8 -fexec-charset=UTF-8)
    endforeach()
    target_link_options(Griffith PRIVATE -municode -static)
    target_link_options(test_griffith_core PRIVATE -static)
endif()
enable_testing()
add_test(NAME physical_consistency COMMAND test_griffith_core)
add_test(NAME native_gui_smoke COMMAND Griffith --smoke-test "${CMAKE_CURRENT_BINARY_DIR}/gui-check")
